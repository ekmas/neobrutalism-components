## Usage

```ts
import { Kbd, KbdGroup } from '@/components/ui/kbd'
```

```tsx
<Kbd>⌘</Kbd>
```

```tsx
<KbdGroup>
  <Kbd>Ctrl</Kbd>
  <Kbd>K</Kbd>
</KbdGroup>
```
