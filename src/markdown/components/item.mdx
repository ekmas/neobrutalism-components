## Usage

```ts
import {
  Item,
  ItemActions,
  ItemContent,
  ItemDescription,
  ItemMedia,
  ItemTitle,
} from '@/components/ui/item'
```

```tsx
<Item>
  <ItemMedia variant="icon">
    <InboxIcon />
  </ItemMedia>
  <ItemContent>
    <ItemTitle>Item</ItemTitle>
    <ItemDescription>Item description</ItemDescription>
  </ItemContent>
  <ItemActions>
    <Button variant="neutral" size="sm">
      Action
    </Button>
  </ItemActions>
</Item>
```
